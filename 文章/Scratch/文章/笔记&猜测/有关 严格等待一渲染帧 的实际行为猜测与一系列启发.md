```js
     * Repeatedly run `sequencer.stepThreads` and filter out
     * inactive threads after each iteration.
     */
    _step () {
        if (this.interpolationEnabled) {
            interpolate.setupInitialState(this);
        }

        if (this.profiler !== null) {
            if (stepProfilerId === -1) {
                stepProfilerId = this.profiler.idByName('Runtime._step');
            }
            this.profiler.start(stepProfilerId);
        }

        // Clean up threads that were told to stop during or since the last step
        this.threads = this.threads.filter(thread => !thread.isKilled);
        this.updateThreadMap();

        // Find all edge-activated hats, and add them to threads to be evaluated.
        for (const hatType in this._hats) {
            if (!Object.prototype.hasOwnProperty.call(this._hats, hatType)) continue;
            const hat = this._hats[hatType];
            if (hat.edgeActivated) {
                this.startHats(hatType);
            }
        }
        this.redrawRequested = false;
        this._pushMonitors();
        if (this.profiler !== null) {
            if (stepThreadsProfilerId === -1) {
                stepThreadsProfilerId = this.profiler.idByName('Sequencer.stepThreads');
            }
            this.profiler.start(stepThreadsProfilerId);
        }
        this.emit(Runtime.BEFORE_EXECUTE);
        const doneThreads = this.sequencer.stepThreads();
        if (this.profiler !== null) {
            this.profiler.stop();
        }
        this.emit(Runtime.AFTER_EXECUTE);
        this._updateGlows(doneThreads);
        // Add done threads so that even if a thread finishes within 1 frame, the green
        // flag will still indicate that a script ran.
        this._emitProjectRunStatus(
            this.threads.length + doneThreads.length -
                this._getMonitorThreadCount([...this.threads, ...doneThreads]));
        // Store threads that completed this iteration for testing and other
        // internal purposes.
        this._lastStepDoneThreads = doneThreads;
        if (this.renderer) {
            // @todo: Only render when this.redrawRequested or clones rendered.
            if (this.profiler !== null) {
                if (rendererDrawProfilerId === -1) {
                    rendererDrawProfilerId = this.profiler.idByName('RenderWebGL.draw');
                }
                this.profiler.start(rendererDrawProfilerId);
            }
            // tw: do not draw if document is hidden or a rAF loop is running
            // Checking for the animation frame loop is more reliable than using
            // interpolationEnabled in some edge cases
            if (!document.hidden && !this.frameLoop._interpolationAnimation) {
                this.renderer.draw();
            }
            if (this.profiler !== null) {
                this.profiler.stop();
            }
        }

        if (this._refreshTargets) {
            this.emit(Runtime.TARGETS_UPDATE, false /* Don't emit project changed */);
            this._refreshTargets = false;
        }

        if (!this._prevMonitorState.equals(this._monitorState)) {
            this.emit(Runtime.MONITORS_UPDATE, this._monitorState);
            this._prevMonitorState = this._monitorState;
        }

        if (this.profiler !== null) {
            this.profiler.stop();
            this.profiler.reportFrames();
        }

        if (this.interpolationEnabled) {
            this._lastStepTime = Date.now();
        }
    }
```

```js


    /**
     * Step through all threads in `this.runtime.threads`, running them in order.
     * @return {Array.<!Thread>} List of inactive threads after stepping.
     */
    //步进线程池（我们伟大的渲染帧！！！）
    stepThreads () {
        // Work time is 75% of the thread stepping interval.
        const WORK_TIME = 0.75 * this.runtime.currentStepTime;
        // For compatibility with Scatch 2, update the millisecond clock
        // on the Runtime once per step (see Interpreter.as in Scratch 2
        // for original use of `currentMSecs`)
        this.runtime.updateCurrentMSecs();
        // Start counting toward WORK_TIME.
        this.timer.start();
        // Count of active threads.
        let numActiveThreads = Infinity;
        // Whether `stepThreads` has run through a full single tick.
        let ranFirstTick = false;
        const doneThreads = [];
        // Conditions for continuing to stepping threads:
        // 1. We must have threads in the list, and some must be active.
        // 2. Time elapsed must be less than WORK_TIME.
        // 3. Either turbo mode, or no redraw has been requested by a primitive.
        //我们伟大的逻辑帧！！！（若干逻辑帧）
        while (this.runtime.threads.length > 0 &&
               numActiveThreads > 0 &&
               this.timer.timeElapsed() < WORK_TIME &&
               (this.runtime.turboMode || !this.runtime.redrawRequested)) {
            if (this.runtime.profiler !== null) {
                if (stepThreadsInnerProfilerId === -1) {
                    stepThreadsInnerProfilerId = this.runtime.profiler.idByName(stepThreadsInnerProfilerFrame);
                }
                this.runtime.profiler.start(stepThreadsInnerProfilerId);
            }

            //我们伟大的逻辑帧！！！（单个逻辑帧）
            numActiveThreads = 0;
            let stoppedThread = false;
            // Attempt to run each thread one time.
            const threads = this.runtime.threads;
            for (let i = 0; i < threads.length; i++) {
                const activeThread = this.activeThread = threads[i];
                // Check if the thread is done so it is not executed.
                if (activeThread.stack.length === 0 ||
                    activeThread.status === Thread.STATUS_DONE) {
                    // Finished with this thread.
                    stoppedThread = true;
                    continue;
                }
                //线程被STATUS_YIELD_TICK，且这是第一个逻辑帧（当!ranFirstTick为true）
                //这证明了帧搁置的时序，所以帧让步是以渲染帧为周期，在第一个逻辑帧内恢复
                //而不是单个逻辑帧
                if (activeThread.status === Thread.STATUS_YIELD_TICK &&
                    !ranFirstTick) {
                    // Clear single-tick yield from the last call of `stepThreads`.
                    activeThread.status = Thread.STATUS_RUNNING;
                }
                //我们伟大的执行线程！！
                //证实了运行时状态和单纯搁置都会导致执行
                if (activeThread.status === Thread.STATUS_RUNNING ||
                    activeThread.status === Thread.STATUS_YIELD) {
                    // Normal-mode thread: step.
                    if (this.runtime.profiler !== null) {
                        if (stepThreadProfilerId === -1) {
                            stepThreadProfilerId = this.runtime.profiler.idByName(stepThreadProfilerFrame);
                        }

                        // Increment the number of times stepThread is called.
                        this.runtime.profiler.increment(stepThreadProfilerId);
                    }
                    this.stepThread(activeThread);
                    activeThread.warpTimer = null;
                }
                if (activeThread.status === Thread.STATUS_RUNNING) {
                    numActiveThreads++;
                }
                // Check if the thread completed while it just stepped to make
                // sure we remove it before the next iteration of all threads.
                if (activeThread.stack.length === 0 ||
                    activeThread.status === Thread.STATUS_DONE) {
                    // Finished with this thread.
                    stoppedThread = true;
                }
            }
            // We successfully ticked once. Prevents running STATUS_YIELD_TICK
            // threads on the next tick.
            ranFirstTick = true;

            if (this.runtime.profiler !== null) {
                this.runtime.profiler.stop();
            }

            //我们伟大的过滤失活线程！！！
            // Filter inactive threads from `this.runtime.threads`.
            if (stoppedThread) {
                let nextActiveThread = 0;
                for (let i = 0; i < this.runtime.threads.length; i++) {
                    const thread = this.runtime.threads[i];
                    if (thread.stack.length !== 0 &&
                        thread.status !== Thread.STATUS_DONE) {
                        this.runtime.threads[nextActiveThread] = thread;
                        nextActiveThread++;
                    } else {
                        this.runtime.threadMap.delete(thread.getId());
                        doneThreads.push(thread);
                    }
                }
                this.runtime.threads.length = nextActiveThread;
            }
        }

        this.activeThread = null;

        return doneThreads;
    }
```

```js
    /**
     * Translates the text in the translate block to the language specified in the menu.
     * @param {object} args - the block arguments.
     * @return {Promise} - a promise that resolves after the response from the translate server.
     */
    getTranslate (args) {
        // If the text contains only digits 0-9 and nothing else, return it without
        // making a request.
        if (/^\d+$/.test(args.WORDS)) return Promise.resolve(args.WORDS);

        // Don't remake the request if we already have the value.
        if (this._lastTextTranslated === args.WORDS &&
            this._lastLangTranslated === args.LANGUAGE) {
            return this._translateResult;
        }

        const lang = this.getLanguageCodeFromArg(args.LANGUAGE);

        let urlBase = `${serverURL}translate?language=`;
        urlBase += lang;
        urlBase += '&text=';
        urlBase += encodeURIComponent(args.WORDS);

        const tempThis = this;
        const translatePromise = fetchWithTimeout(urlBase, {}, serverTimeoutMs)
            .then(response => response.text())
            .then(responseText => {
                const translated = JSON.parse(responseText).result;
                tempThis._translateResult = translated;
                // Cache what we just translated so we don't keep making the
                // same call over and over.
                tempThis._lastTextTranslated = args.WORDS;
                tempThis._lastLangTranslated = args.LANGUAGE;
                return translated;
            })
            .catch(err => {
                log.warn(`error fetching translate result! ${err}`);
                return args.WORDS;
            });
        return translatePromise;
    }
```

来自CCW的一条这条评论让我发现了和之前只有一个准绳的 真·等待一帧（即翻译积木） 匹敌的其他原版积木
（翻译积木的那行`if (/^\d+$/.test(args.WORDS)) return Promise.resolve(args.WORDS);
`是关键）
即 [播放并等待]（在特定准备的条件下）与[说()(0)秒]（忽略不能正常使用[说][思考]等积木的缺点））。
我大胆猜测：**只要是微任务而且立刻callback就可以严格逐渲染帧推迟、搁置与精确调度。**
因为结合`sequencer.js`和`runtime.js`的源码，
那个stepThreads（注意这是单次渲染帧本帧）是同步函数
再结合专门针对异步的 挂起行为 和 线程状态 。
我大胆猜是 **js的事件循环诱导微任务精确**
（因为明显同步函数里不可能await一个异步对象，否则整个链都应当是异步函数，因此只能是靠Scratch内核的让步机制挂起异步线程，这就引入了宏任务和微任务。）
（因为只在stepTheads之后只会渲染一次（单个渲染帧的关键的定义处））。
若干stepTheads调用的间隙，经过事件循环后微任务返回（因为明显不可能在一次渲染之内连续调用多个stepTheads，也不可能渲染无效导致不进入微任务），
所以再一次stepTheads（即渲染帧率）的调度是严格满足 真·等待一帧 的基本应有的性质。
**所以 严格等待渲染帧 的调度可行性条件可能非常松？！**
我再大胆猜一下：**只要是涉及等待的积木，且靠异步写等待逻辑的回调函数的积木函数，只要可干涉内部的判定条件，使得其在异步对象里立刻callback，或者直接返回一个立刻callback的异步对象的写法，可能就都是精确的等待一帧**

还有一个理由：
那个翻译积木就是直接callback的。所以其他可直接callback的（比如`sayforsecs`的`return new Promise(resolve => { ... }, 1000 * args.SECS);`），也应当和翻译积木（即真·等待一帧）匹敌。（如果忽略原语义和缺点的话）

虽然有点本末倒置，因为我疑似在拿一个积木反过来定义渲染帧。
但是实际上结合前面的宏任务和微任务的事件循环，又因为 **宏任务->微任务->渲染** 的web底层
（明显TurboWarp那么大的体量几乎必然会产生渲染。该调度链总是一定，尽管渲染在web中是可选的）
那么我就可以解释为什么宏任务的stepThread加上微任务立刻回调必然是严格等待渲染帧的。

因为根据“宏任务->微任务->渲染”的调度序，真·等待一帧的完整过程可能是这样发生的：

1. 执行 stepThreads 中：在这个同步的宏任务期间，Scratch的虚拟机开始处理某个“逻辑帧”上的线程，执行到了一个积木（如 翻译积木）。
2. 积木函数内部返回的异步对象设置立即回调，return之后被虚拟机检测到异步对象被异步搁置（解释模式：`Scratch-vm\src\engine\execute.js`，编译模式不详但是应该一致），继续遍历线程池：积木将返回一个 Promise 异步对象时，虚拟机会向微任务队列中添加一个微任务；此后一旦当任务被调度时，必然callback，将会返回结果
3. stepThreads 执行完毕：宏任务结束。
4. 清空微任务队列：事件循环开始处理微任务，调度那个 Promise 异步对象的 resolve 回调，将之前被异步搁置的Scratch线程状态恢复为“可运行”（或者调度时检查异步对象状态？这部分我不太确定。总之大概就是能恢复。）。
- 尝试渲染：清空微任务队列后，浏览器将有机会进行渲染、刷新屏幕，对应于“渲染帧”的渲染操作。
- 下一次事件循环迭代：新一轮事件循环开始，会执行下一个宏任务，在下一次 stepThreads 被执行时，那个原先被异步搁置的线程现在由于异步对象callback了，所以再次被调度时必然恢复、正常执行。

之所以说这个，是因为渲染帧的判定是看 渲染时机 来定义的。不可能用微任务和宏任务定义，所以必须思考渲染到底在哪里。或者结合web实现，这几乎指向同一事实。

不过这并不意味着 翻译积木可直接被替代 或者 真·等待一帧 很普遍。
明显声音类、外观类积木有对应的职能和语义，没有“翻译积木的无副作用的特殊行为”那么纯。
因此很长时间我仍然使用 翻译积木传递特殊参数 作为主要写法，
但也不意味着仔细分析这些“不纯”的等帧的 性能开销 和 适用情况 和 缺点 没有意义，
我觉得把 js事件循环、渲染帧与逻辑帧与线程池与线程与线程行为与线程状态与调度 全部串起来可太有意思了。

当然得声明一下：
一个纯粹的“真·等待一帧”积木，**不能有标记“渲染标记”的副作用**，而应 **有且仅负责搁置线程等待，直到下一个渲染帧的边界部分被调度时才恢复执行**。
因为“真·等待一帧”通过微任务异步回调机制，确保回调立刻发生在微任务阶段，
而实际执行要等到下一次stepThreads（即下一个宏任务），
这天然地保证了线程在下一个渲染帧边界恢复，而不依赖redrawRequested（渲染标记），
总之，这是由事件循环时序保证的（目前看来），而不是由渲染标记驱动的

否则 [等待0秒] 等依赖调度机制的，**仅等待一个逻辑帧**的，又因为**标记渲染标记**的*奇怪调度行为* 的组合，会在一些环境（如LRF，**可LRF甚至很普遍**）中被错误的视为“等待一渲染帧”
（嗯对，“真·等待一帧”始终是“真·等待一渲染帧”的简称，而不是逐逻辑帧）
